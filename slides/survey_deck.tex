    Covariates & Poverty, median income, prior uninsurance, share $<$18, share $>$65 \\

  \end{center}
  \vspace{0.3cm}
  {\footnotesize\color{WarmGray} The 2015--2017 cohorts are dropped because they contain 3, 3 and 1 states, and at that size the estimator's variance is not reliably identified: in placebo runs a one-state cohort produces a falsely significant ATT 40\% of the time and fails the pre-trend test 88.5\% of the time. With them excluded there is a single treatment date, so this is a standard two-group design.}
\end{frame}

    \toprule

    Price per unit & spend / volume \\
    Volume per enrollee & volume / enrollees \\
    Spend per enrollee & price $\times$ volume per enrollee \\
    Spend per capita & spend per enrollee $\times$ coverage \\
    Volume per capita & volume per enrollee $\times$ coverage \\
    \bottomrule
  \end{tabular}}

  {\footnotesize\color{WarmGray} Payers: Medicaid, Medicare, private, out-of-pocket, and their sum. Settings: all care plus ambulatory, emergency, inpatient, nursing facility and pharmacy.}

\begin{frame}{Candidates for the poster}
  \vspace{-0.2cm}
  \openq{\textbf{First stage.} Medicaid coverage $+3.7$pp, private $-2.2$pp. Both pass the pre-test.}
  \vspace{0.02cm}
  \openq{\textbf{Headline.} Total spending $+\$206$ per capita, Medicaid $+\$189$, no offset elsewhere.}
  \vspace{0.02cm}
  \openq{\textbf{The mechanism.} Spending per enrollee is flat while coverage rises: the per-capita effect is an enrollment effect.}
  \vspace{0.02cm}
  \openq{\textbf{Setting split.} Ambulatory and inpatient carry the aggregate; pharmacy and nursing facility are flat.}
  \vspace{0.15cm}
  {\scriptsize\color{WarmGray} Leave off: price per unit, which fails the pre-test for Medicaid, and volume per enrollee, whose negative coefficient needs more explanation than a poster allows.}
\end{frame}
