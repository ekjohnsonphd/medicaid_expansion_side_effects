\transitionslide{Open}{Choices not yet made}

\begin{frame}{Four decisions remain}
  \vspace{-0.1cm}
  \openq{\textbf{Primary sample.} The pre-trend test is only interpretable on the 2014 cohort. Making it primary trades 7 treated states for a testable assumption.}
  \vspace{0.1cm}
  \openq{\textbf{Out-of-pocket.} Every per-capita model fails the pre-test. Report as unidentified, drop, or change design.}
  \vspace{0.1cm}
  \openq{\textbf{Louisiana.} Alone in the 2017 cohort with two post-periods. Currently retained.}
  \vspace{0.1cm}
  \openq{\textbf{Waiver states.} The \texttt{restricted} sensitivity is coded but not yet run across the matrix.}
\end{frame}
