\begin{frame}{Outcomes are the terms of one accounting identity}
  \vspace{0.75cm}
  \begin{center}
  {\Large\color{DeepNavy}
  $\underbrace{\dfrac{\text{spend}}{\text{pop}}}_{\text{per capita}}
   =
   \underbrace{\dfrac{\text{spend}}{\text{vol}}}_{\text{price}}
   \times
   \underbrace{\dfrac{\text{vol}}{\text{enrollees}}}_{\text{use per enrollee}}
   \times
   \underbrace{\dfrac{\text{enrollees}}{\text{pop}}}_{\text{coverage}}$}
  \end{center}
  \vspace{0.75cm}
  {\footnotesize\color{WarmGray} Enrollees and population are recovered from the DEX ratios and reconcile exactly on the \textit{All ages} basis. Each term is estimated separately below, so the channels can be read against one another rather than inferred.}
\end{frame}
